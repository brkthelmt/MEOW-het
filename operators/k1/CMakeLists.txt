cmake_minimum_required(VERSION 3.18)
project(meow_k1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SOC_VERSION "Ascend910B2" CACHE STRING "Ascend SoC")
set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN")
include(${MEOW_ASCENDC_CMAKE})
ascendc_library(meow_k1_kernel STATIC k1_queue.cpp k1_manual.cpp)
ascendc_compile_options(meow_k1_kernel PRIVATE -ffp-contract=off)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import torch;print(torch.utils.cmake_prefix_path)" OUTPUT_VARIABLE TORCH_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${TORCH_PREFIX}")
find_package(Torch REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import importlib.util,pathlib;print(pathlib.Path(importlib.util.find_spec('torch_npu').origin).parent)" OUTPUT_VARIABLE NPU_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
add_library(meow_k1 SHARED binding.cpp)
target_compile_options(meow_k1 PRIVATE ${TORCH_CXX_FLAGS})
target_include_directories(meow_k1 PRIVATE ${NPU_ROOT}/include ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(meow_k1 PRIVATE ${NPU_ROOT}/lib ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(meow_k1 PRIVATE meow_k1_kernel ${TORCH_LIBRARIES} torch_npu ascendcl)
